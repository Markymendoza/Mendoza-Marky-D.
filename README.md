def chapter7():

    print("\n====================================================\n")
    print("Iteration allows a program to perform the same task multiple times.")
    print("Without a loop, instructions must be written repeatedly.\n")
    print("EXAMPLE:\n")
    print("Without a loop:")
    print("print(\"Study Python\")")
    print("print(\"Study Python\")")
    print("print(\"Study Python\")")
    print("print(\"Study Python\")\n")
    print("Using a loop:")
    print("for number in range(4):")
    print("    print(\"Study Python\")\n")
    print("Both programs produce the same output, but the loop makes the program shorter and")
    print("easier to modify. To repeat the message 100 times, change 4 to 100.")
    print("\n-------------------------------------------------------------------------------------\n")
    pause()

    print("\n-------------------------------------------------------------------------------------\n")
    print("TYPES OF LOOPS IN PYTHON")
    print("Python has two main types of loops: for and while.\n")
    print("1. for loop:")
    print("   - Repeats instructions for each item in a sequence.")
    print("   - Useful when the number of repetitions is known.")
    print("\nEXAMPLE:\n")
    print("              for number in range(3):")
    print("                  print(\"Hello!\")\n")
    print("\n-------------------------------------------------------------------------------------\n")
    print("2. while loop:")
    print("   - Repeats instructions while a condition is True.")
    print("   - Useful when repetition depends on a condition.")
    print("\nEXAMPLE:\n")
    print("              count = 1;")
    print("              while count <= 3:")
    print("                  print(\"Hello!\")")
    print("                  count += 1\n")
    print("Both examples print Hello! three times.")
    print("\n-------------------------------------------------------------------------------------\n")
    pause()

    print("\n-------------------------------------------------------------------------------------\n")
    print("PYTHON FOR LOOP")
    print("\nSYNTAX:")
    print("for variable in sequence:")
    print("    statement(s)\n")
    print("variable: receives each item")
    print("sequence: items to loop through")
    print("Indented statements: the loop body\n")
    print("\n-------------------------------------------------------------------------------------\n")
    pause()

    print("\n-------------------------------------------------------------------------------------\n")
    print("PYTHON WHILE LOOP")
    print("SYNTAX:")
    print("while condition:")
    print("    statement(s)\n")
    print("condition: checked before each iteration")
    print("Indented statements: run while the condition is True.\n")
    print("Example Code:")
    print("count = 1")
    print("while count <= 5:")
    print("    print(count)")
    print("    count += 1\n")
    print("count += 1 increases count by 1 each iteration.")
